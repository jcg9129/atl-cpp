cmake_minimum_required(VERSION 3.25)
project(atl_consumer LANGUAGES CXX)
find_package(atl 0.1 CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE atl::atl)
# Check installed headers with the same MSVC warning policy as project builds.
option(DISABLE_WERROR "Do not treat project warnings as errors" OFF)
if(MSVC)
    target_compile_options(consumer PRIVATE /W4 /permissive- /Zc:__cplusplus /Zc:preprocessor /utf-8)
    if(NOT DISABLE_WERROR)
        target_compile_options(consumer PRIVATE /WX)
    endif()
endif()
enable_testing()
add_test(NAME consumer COMMAND consumer)
